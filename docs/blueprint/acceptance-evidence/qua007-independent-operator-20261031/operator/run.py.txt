import subprocess,sys,pathlib,time,json,os
root=pathlib.Path(__file__).resolve().parent
name=sys.argv[1]; cmd=sys.argv[2]; cwd=sys.argv[3]
(root/(name+'.command')).write_text(cmd+'\n')
t=time.time(); p=subprocess.run(['bash','-c',cmd],cwd=cwd,stdout=subprocess.PIPE,stderr=subprocess.PIPE)
secret=os.environ.get('OPENCLAW_GATEWAY_TOKEN','').encode()
for ext,data in [('stdout',p.stdout),('stderr',p.stderr)]:
 if secret: data=data.replace(secret,b'[REDACTED]')
 (root/(name+'.'+ext)).write_bytes(data)
(root/(name+'.result.json')).write_text(json.dumps({'exit':p.returncode,'start':t,'elapsed':time.time()-t,'cwd':cwd})+'\n')
print(name,p.returncode,round(time.time()-t,2));print(p.stdout.decode(errors='replace')[-2000:]);print(p.stderr.decode(errors='replace')[-2000:]);sys.exit(p.returncode)
