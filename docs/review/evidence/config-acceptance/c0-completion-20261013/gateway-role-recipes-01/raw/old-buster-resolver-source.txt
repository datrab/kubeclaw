function resolveContext(ref,capture,expected){assert(validContextRef(ref),'CONTEXT_REF_CLOSED_SCHEMA:'+JSON.stringify(validContextRef.errors));assert(ref.path==='managed-bootstrap-context.json','CONTEXT_WRONG_PATH');assert(contextSha(capture)===ref.sha256,'CONTEXT_SOURCE_SHA');assert(validContextCapture(capture),'CONTEXT_CAPTURE_SCHEMA:'+JSON.stringify(validContextCapture.errors));assert(eq(Object.keys(capture).sort(),['contexts','nativeFullSchema']),'CONTEXT_CAPTURE_CLOSED');const c=pointer(capture,ref.pointer);assert(validAssociation(c.association),'CONTEXT_ASSOCIATION_SCHEMA:'+JSON.stringify(validAssociation.errors));assert(ref.pointer==='/contexts/'+c.association.receivingProfileId,'CONTEXT_POINTER_PROFILE_ASSOCIATION');assert(capture.nativeFullSchema===contextCapture.nativeFullSchema,'CONTEXT_NATIVE_SCHEMA_SOURCE');for(const [key,value]of Object.entries(expected))assert(c.association[key]===value,'CONTEXT_ASSOCIATION_'+key);assert(c.agentRole===c.association.role,'CONTEXT_RECIPE_ROLE');const recipe=structuredClone(c);delete recipe.association;assert(eq(recipe,recipes[c.association.sourceRecipeId]),'CONTEXT_READONLY_SOURCE_RECIPE');return c;}
